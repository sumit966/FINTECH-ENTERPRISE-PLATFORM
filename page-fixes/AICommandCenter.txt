import { useEffect, useRef, useState } from 'react';
import { Send, Sparkles, AlertTriangle, User } from 'lucide-react';
import { Card } from '../../components/ui/Card';
import Button from '../../components/ui/Button';
import Badge from '../../components/ui/Badge';
import { aiService, AiAnswer } from '../../services/aiService';

interface ChatTurn {
  id: string;
  role: 'user' | 'assistant';
  text: string;
  answer?: AiAnswer;
}

const SUGGESTIONS = [
  'backend engineers in Pune',
  'which projects are delayed',
  'who is on PRJ-021',
  'top clients by revenue',
  'employees with AWS skill',
  'how many employees do we have',
  'show me our revenue',
  'who am I',
];

export default function AICommandCenter() {
  const [provider, setProvider] = useState<string>('');
  const [turns, setTurns] = useState<ChatTurn[]>([]);
  const [input, setInput] = useState('');
  const [busy, setBusy] = useState(false);
  const endRef = useRef<HTMLDivElement>(null);

  useEffect(() => { aiService.provider().then((r) => setProvider(r.provider)).catch(() => {}); }, []);
  useEffect(() => { endRef.current?.scrollIntoView({ behavior: 'smooth' }); }, [turns]);

  const send = async (q: string) => {
    if (!q.trim() || busy) return;
    const userTurn: ChatTurn = { id: 'u' + Date.now(), role: 'user', text: q };
    setTurns((t) => [...t, userTurn]);
    setInput('');
    setBusy(true);
    try {
      const res = await aiService.ask(q);
      setTurns((t) => [...t, { id: 'a' + Date.now(), role: 'assistant', text: res.answer, answer: res }]);
    } catch (e: any) {
      setTurns((t) => [...t, { id: 'a' + Date.now(), role: 'assistant', text: 'Error: ' + (e?.message || 'request failed') }]);
    } finally { setBusy(false); }
  };

  return (
    <div className="space-y-6">
      <div className="flex items-start justify-between gap-4 flex-wrap">
        <div>
          <h1 className="text-2xl font-semibold tracking-tight">AI Command Center</h1>
          <p className="text-sm text-slate-500 dark:text-slate-400 mt-1">
            Natural language queries over your enterprise data, scoped by your RBAC permissions
          </p>
        </div>
        <div className="inline-flex items-center gap-2 rounded-md border border-brand-500/40 bg-brand-500/10 px-3 py-1.5 text-xs">
          <Sparkles size={14} className="text-brand-500" />
          <span className="font-medium">Provider: {provider || '…'}</span>
        </div>
      </div>

      {provider === 'rule-based' && (
        <Card className="border-amber-500/30 bg-amber-500/5">
          <div className="flex items-start gap-2 text-xs text-amber-700 dark:text-amber-300">
            <AlertTriangle size={14} className="mt-0.5 shrink-0" />
            <div>
              <strong>Deterministic query engine &mdash; no external LLM configured.</strong> Answers are generated by parsing your question and running permission-scoped database queries. To enable an LLM, set <code className="font-mono">AI_PROVIDER=openai</code> and <code className="font-mono">AI_API_KEY</code> in <code className="font-mono">backend/.env</code>. Until then, no data leaves this server.
            </div>
          </div>
        </Card>
      )}

      <Card className="min-h-[500px] flex flex-col" padded={false}>
        <div className="flex-1 overflow-y-auto p-5 space-y-5">
          {turns.length === 0 && (
            <div className="text-center py-12">
              <Sparkles size={28} className="mx-auto text-brand-500 mb-3" />
              <h3 className="font-semibold">Ask anything about the enterprise</h3>
              <p className="text-sm text-slate-500 mt-1">Try one of these:</p>
              <div className="mt-4 flex flex-wrap gap-2 justify-center max-w-2xl mx-auto">
                {SUGGESTIONS.map((s) => (
                  <button key={s} onClick={() => send(s)} className="rounded-full border border-slate-200 dark:border-slate-700 px-3 py-1.5 text-xs hover:border-brand-500 hover:text-brand-500 transition">
                    {s}
                  </button>
                ))}
              </div>
            </div>
          )}

          {turns.map((t) => (
            <div key={t.id} className={'flex gap-3 ' + (t.role === 'user' ? 'justify-end' : '')}>
              {t.role === 'assistant' && (
                <div className="h-8 w-8 rounded-full bg-brand-500/15 grid place-items-center shrink-0">
                  <Sparkles size={14} className="text-brand-500" />
                </div>
              )}
              <div className={'max-w-2xl ' + (t.role === 'user' ? 'text-right' : '')}>
                {t.role === 'user' ? (
                  <div className="inline-block rounded-2xl bg-brand-500 text-white px-4 py-2 text-sm">{t.text}</div>
                ) : (
                  <div className="space-y-3">
                    <div className="rounded-2xl border border-slate-200 dark:border-slate-800 bg-white dark:bg-slate-900 px-4 py-3 text-sm">
                      {t.text}
                    </div>
                    {t.answer?.rows && t.answer.rows.length > 0 && (
                      <div className="rounded-xl border border-slate-200 dark:border-slate-800 overflow-hidden">
                        <table className="w-full text-xs">
                          <thead className="bg-slate-50 dark:bg-slate-900/60">
                            <tr>
                              {Object.keys(t.answer.rows[0]).map((k) => (
                                <th key={k} className="text-left px-3 py-2 font-medium uppercase tracking-wider text-slate-500">{k}</th>
                              ))}
                            </tr>
                          </thead>
                          <tbody className="divide-y divide-slate-200 dark:divide-slate-800">
                            {t.answer.rows.slice(0, 15).map((r, i) => (
                              <tr key={i}>
                                {Object.values(r).map((v, j) => (
                                  <td key={j} className="px-3 py-2">{String(v)}</td>
                                ))}
                              </tr>
                            ))}
                          </tbody>
                        </table>
                        {t.answer.rows.length > 15 && (
                          <div className="px-3 py-2 text-[11px] text-slate-500 bg-slate-50 dark:bg-slate-900/60">
                            + {t.answer.rows.length - 15} more rows
                          </div>
                        )}
                      </div>
                    )}
                    {t.answer && (
                      <div className="text-[10px] text-slate-400">
                        answered by <Badge tone="brand">{t.answer.provider}</Badge>
                      </div>
                    )}
                  </div>
                )}
              </div>
              {t.role === 'user' && (
                <div className="h-8 w-8 rounded-full bg-slate-200 dark:bg-slate-800 grid place-items-center shrink-0">
                  <User size={14} className="text-slate-500" />
                </div>
              )}
            </div>
          ))}
          {busy && (
            <div className="flex items-center gap-2 text-sm text-slate-500">
              <span className="inline-block h-2 w-2 rounded-full bg-brand-500 animate-pulse" />
              Thinking…
            </div>
          )}
          <div ref={endRef} />
        </div>

        <div className="border-t border-slate-200 dark:border-slate-800 p-4">
          <form onSubmit={(e) => { e.preventDefault(); send(input); }} className="flex gap-2">
            <input
              value={input}
              onChange={(e) => setInput(e.target.value)}
              placeholder="Ask a question… e.g. 'show backend engineers in Pune'"
              className="flex-1 rounded-md border border-slate-200 dark:border-slate-700 bg-white dark:bg-slate-900 px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-brand-500/40"
            />
            <Button type="submit" disabled={busy || !input.trim()}>
              <Send size={14} /> Send
            </Button>
          </form>
        </div>
      </Card>
    </div>
  );
}
